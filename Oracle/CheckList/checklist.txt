    '<tr>' ||
    '<td>' || rs.inst_id || '</td>' ||
    '<td>' || rs.resource_name || '</td>' ||
    '<td>' || rs.current_utilization || '</td>' ||
    '<td>' || rs.max_utilization || '</td>' ||
    '<td>' || par.value || '</td>' ||
    '</tr>'

  AND rs.inst_id = par.inst_id
ORDER BY rs.resource_name, rs.inst_id;

check_tablespace() {
if [[ $DB_VERSION_NUMERIC -eq 11 ]] || [[ $CHECK_PDB_NUMERIC -eq 0 ]] ; then  
    {
        echo "<h3>Tablespaces (Permanent & Temporary)</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>TABLESPACE_NAME</th><th>ALLOCATED_GB</th><th>FREE_GB</th><th>USED_GB</th><th>USED_PCT</th><th>MAX_BYTES_GB</th></tr>"
    } >> "$HTML_FILE"

    sqlplus -s /@$ORACLE_SID >> "$HTML_FILE" <<EOF
SET HEADING OFF
SET FEEDBACK OFF
SET PAGESIZE 0
SET LINESIZE 500
SET TERMOUT OFF
SET ECHO OFF

SELECT '<tr>'
|| '<td>' || Tablespace_Name || '</td>'
|| '<td>' || ROUND(Allocated_MB / 1024,2) || '</td>'
|| '<td>' || ROUND(Free_MB / 1024,2) || '</td>'
|| '<td>' || ROUND(Used_MB / 1024,2) || '</td>'
|| '<td class="' ||
CASE 
    WHEN used_pct >= 90 THEN 'critico'
    WHEN used_pct >= 70 THEN 'alerta'
    ELSE 'ok'
END || '">' || used_pct || '%</td>'
|| '<td>' || ROUND(Max_Bytes_MB / 1024,2) || '</td>'
|| '</tr>'
FROM (
    SELECT  a.tablespace_name AS Tablespace_Name,
            ROUND(a.bytes_alloc / 1024 / 1024) Allocated_MB,
            ROUND(NVL(b.bytes_free, 0) / 1024 / 1024) Free_MB,
            ROUND((a.bytes_alloc - NVL(b.bytes_free, 0)) / 1024 / 1024) Used_MB,
            ROUND((NVL(b.bytes_free, 0) / a.bytes_alloc) * 100) free_pct,
            100 - ROUND((NVL(b.bytes_free, 0) / a.bytes_alloc) * 100) used_pct,
            ROUND(maxbytes / 1024 / 1024) Max_Bytes_MB
    FROM  ( SELECT f.tablespace_name,
                   SUM(f.bytes) bytes_alloc,
                   SUM(DECODE(f.autoextensible, 'YES', f.maxbytes, 'NO', f.bytes)) maxbytes
            FROM dba_data_files f
            GROUP BY tablespace_name) a,
          ( SELECT f.tablespace_name,
                   SUM(f.bytes) bytes_free
            FROM dba_free_space f
            GROUP BY tablespace_name) b
    WHERE a.tablespace_name = b.tablespace_name (+)
    UNION ALL
    SELECT h.tablespace_name AS tablespace_name,
           ROUND(SUM(h.bytes_free + h.bytes_used) / 1048576) megs_alloc,
           ROUND(SUM((h.bytes_free + h.bytes_used) - NVL(p.bytes_used, 0)) / 1048576) megs_free,
           ROUND(SUM(NVL(p.bytes_used, 0)) / 1048576) megs_used,
           ROUND((SUM((h.bytes_free + h.bytes_used) - NVL(p.bytes_used, 0)) / SUM(h.bytes_used + h.bytes_free)) * 100) pct_free,
           100 - ROUND((SUM((h.bytes_free + h.bytes_used) - NVL(p.bytes_used, 0)) / SUM(h.bytes_used + h.bytes_free)) * 100) pct_used,
           ROUND(SUM(f.maxbytes) / 1048576) max
    FROM   (SELECT DISTINCT * FROM sys.gv_\$TEMP_SPACE_HEADER) h,
           (SELECT DISTINCT * FROM sys.gv_\$Temp_extent_pool) p,
           dba_temp_files f
    WHERE  p.file_id(+) = h.file_id
    AND    p.tablespace_name(+) = h.tablespace_name
    AND    f.file_id = h.file_id
    AND    f.tablespace_name = h.tablespace_name
    GROUP BY h.tablespace_name
    ORDER BY 2
) sub1
ORDER BY used_pct DESC;

EXIT;
EOF

    echo "</table>" >> "$HTML_FILE"
else
    {
        echo "<h3>Tablespaces (Permanent & Temporary)</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>NAME</th><th>TABLESPACE_NAME</th><th>ALLOCATED_GB</th><th>USED_GB</th><th>USED_PCT</th><th>MAX_BYTES_GB</th></tr>"
    } >> "$HTML_FILE"

    sqlplus -s /@$ORACLE_SID >> "$HTML_FILE" <<EOF
SET HEADING OFF
SET FEEDBACK OFF
SET PAGESIZE 0
SET LINESIZE 500
SET TERMOUT OFF
SET ECHO OFF

SELECT 
    '<tr>' ||
    '<td>' || c.name || '</td>' ||
    '<td>' || a.tablespace_name || '</td>' ||
    '<td>' || ROUND(NVL(a.physical_bytes,0)/(1024*1024*1024),2) || '</td>' ||
    '<td>' || ROUND(NVL(b.tot_used,0)/(1024*1024*1024),2) || '</td>' ||
    '<td class="' ||
        CASE
            WHEN a.physical_bytes > 0 AND ROUND((NVL(b.tot_used,0)/a.physical_bytes)*100,2) >= 90 THEN 'critico'
            WHEN a.physical_bytes > 0 AND ROUND((NVL(b.tot_used,0)/a.physical_bytes)*100,2) >= 70 THEN 'alerta'
            ELSE 'ok'
        END || '">' || 
        CASE 
            WHEN a.physical_bytes > 0 THEN ROUND((NVL(b.tot_used,0)/a.physical_bytes)*100,2) || '%'
            ELSE '0%'
        END || '</td>' ||
    '<td>' || ROUND(a.bytes_alloc/(1024*1024*1024),2) || '</td>' ||
    '</tr>' AS html_row
FROM (
    SELECT con_id, tablespace_name,
           SUM(bytes) physical_bytes,
           SUM(DECODE(autoextensible,'NO',bytes,'YES',maxbytes)) bytes_alloc
    FROM cdb_data_files
    GROUP BY con_id, tablespace_name
) a
LEFT JOIN (
    SELECT con_id, tablespace_name, SUM(bytes) tot_used
    FROM cdb_segments
    GROUP BY con_id, tablespace_name
) b
ON a.con_id = b.con_id AND a.tablespace_name = b.tablespace_name
JOIN (
    SELECT name, con_id FROM v\$containers
) c
ON a.con_id = c.con_id
ORDER BY c.name, ROUND((NVL(b.tot_used,0)/NULLIF(a.physical_bytes,0))*100,2) DESC;

exit;

EOF
    echo "</table>" >> "$HTML_FILE"

fi

}

check_dataguard() {
    {
        echo "<h3>Data Guard Status</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>DATABASE_ROLE</th><th>PROTECTION_MODE</th><th>SWITCHOVER_STATUS</th><th>NAME</th><th>THREAD</th><th>STANDBY_LAST_RECEIVED</th><th>STANDBY_LAST_APPLIED</th><th>STANDBY_DT_LAST_APP</th><th>MINUTOS</th><th>ARC_DIFF</th></tr>"
    } >> "$HTML_FILE"

    sqlplus -s /@$ORACLE_SID >> "$HTML_FILE" <<EOF
SET HEADING OFF
SET FEEDBACK OFF
SET PAGESIZE 0
SET LINESIZE 1000
SET TERMOUT OFF
SET ECHO OFF


SELECT '<tr>'
|| '<td>' || DATABASE_ROLE || '</td>'
|| '<td>' || PROTECTION_MODE || '</td>'
|| '<td>' || SWITCHOVER_STATUS || '</td>'
|| '<td>' || NAME || '</td>'
|| '<td>' || THREAD || '</td>'
|| '<td>' || STANDBY_LAST_RECEIVED || '</td>'
|| '<td>' || STANDBY_LAST_APPLIED || '</td>'
|| '<td>' || STANDBY_DT_LAST_APP || '</td>'
|| '<td>' || MINUTOS || '</td>'
|| '<td class="' ||
CASE
    WHEN ARC_DIFF > 8 THEN 'critico'
    WHEN ARC_DIFF > 3 THEN 'alerta'
    ELSE 'ok'
END || '">' || ARC_DIFF || '</td>'
|| '</tr>'
FROM (
    SELECT CASE WHEN ARC_DIFF <= 3 THEN ':)' WHEN ARC_DIFF > 3 AND ARC_DIFF <= 8 THEN ':|' ELSE ':(' END ST,
           z.DATABASE_ROLE, z.PROTECTION_MODE, z.SWITCHOVER_STATUS, z.name, z.thread,
           z.STANDBY_LAST_RECEIVED, z.STANDBY_LAST_APPLIED, z.STANDBY_DT_LAST_APP,
           z.MINUTOS, z.ARC_DIFF
    FROM (
        SELECT c.DATABASE_ROLE,
               c.PROTECTION_MODE,
               c.SWITCHOVER_STATUS,
               a.DEST_ID,
               (SELECT MAX(NVL2(xx.name, xx.DEST_ID || ' - ' || xx.name, NULL))
                FROM v\$archived_log xx
                WHERE xx.DEST_ID = a.DEST_ID
                  AND xx.resetlogs_change# = (SELECT resetlogs_change# FROM v\$database)
                  AND SEQUENCE# = (SELECT MAX(yy.SEQUENCE#)
                                   FROM v\$archived_log yy
                                   WHERE yy.resetlogs_change# = (SELECT resetlogs_change# FROM v\$database)
                                     AND yy.DEST_ID = xx.DEST_ID)) AS name,
               a.thread# thread,
               b.last_seq STANDBY_LAST_RECEIVED,
               a.applied_seq STANDBY_LAST_APPLIED,
               TO_CHAR(a.last_app_timestamp,'DD/MM/YYYY HH24:MI:SS') AS STANDBY_DT_LAST_APP,
               (SYSDATE - a.last_app_timestamp) * 24 * 60 AS MINUTOS,
               b.last_seq - a.applied_seq ARC_DIFF
        FROM (SELECT DEST_ID, thread#,
                     MAX(sequence#) applied_seq,
                     MAX(next_time) last_app_timestamp
              FROM gv\$archived_log
              WHERE applied = 'YES'
                AND resetlogs_change# = (SELECT resetlogs_change# FROM v\$database)
              GROUP BY DEST_ID, thread#) a,
             (SELECT DEST_ID, thread#,
                     MAX(sequence#) last_seq
              FROM gv\$archived_log
              WHERE resetlogs_change# = (SELECT resetlogs_change# FROM v\$database)
              GROUP BY DEST_ID, thread#) b,
             (SELECT DATABASE_ROLE, PROTECTION_MODE, SWITCHOVER_STATUS FROM v\$database) c
        WHERE a.thread# = b.thread#
          AND a.DEST_ID = b.DEST_ID
    ) z
    WHERE UPPER(z.NAME) NOT LIKE '%ARCHIVELOG%'
      AND ((z.DATABASE_ROLE = 'PRIMARY' AND (z.thread, z.DEST_ID) IN
            (SELECT INST_ID thread#, DEST_ID FROM gv\$archive_dest_status WHERE STATUS <> 'INACTIVE'))
           OR (z.DATABASE_ROLE <> 'PRIMARY'))
    ORDER BY z.name, z.thread
);

EXIT;
EOF

    echo "</table>" >> "$HTML_FILE"
}



check_session() {
    {
        echo "<h3>Sessões Ativas</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>INST_ID</th><th>SID</th><th>SERIAL#</th><th>USERNAME</th><th>MACHINE</th><th>SQL_ID</th><th>ELAPSED_TIME</th><th>STATUS</th><th>EVENT</th><th>SERVICE_NAME</th></tr>"
    } >> "$HTML_FILE"

    sqlplus -s /@$ORACLE_SID >> "$HTML_FILE" <<EOF
SET HEADING OFF
SET FEEDBACK OFF
SET PAGESIZE 0
SET LINESIZE 1000
SET TERMOUT OFF
SET ECHO OFF

SELECT
    '<tr>' ||
    '<td>' || inst_id || '</td>' ||
    '<td>' || sid || '</td>' ||
    '<td>' || serial# || '</td>' ||
    '<td>' || username || '</td>' ||
    '<td>' || machine || '</td>' ||
    '<td>' || sql_id || '</td>' ||
    '<td>' || elapsed_time || '</td>' ||
    '<td>' || status || '</td>' ||
    '<td>' || event || '</td>' ||
    '<td>' || service_name || '</td>' ||
    '</tr>'
FROM (
select distinct inst_id, sid, serial#, username, machine, sql_id, elapsed_time, status, event,service_name from (
select s.inst_id, s.sid, s.serial#, s.username, s.machine, gv.sql_text, gv.sql_id,
TO_CHAR(TRUNC(s.last_call_et/3600),'FM9900') || ':' || TO_CHAR(TRUNC(MOD(s.last_call_et,3600)/60),'FM00') || ':' || TO_CHAR(MOD(s.last_call_et,60),'FM00') as elapsed_time,
s.status, s.event,s.SERVICE_NAME
from gv\$sql gv, gv\$session s
where gv.sql_id=s.sql_id
and gv.inst_id=s.inst_id
and username is not null
and s.status='ACTIVE'
)
) order by elapsed_time desc;

EXIT;
EOF

    echo "</table>" >> "$HTML_FILE"
}

        echo "<tr><th>OWNER</th><th>SEQUENCE_NAME</th><th>MAX_VALUE</th><th>INCREMENT_BY</th><th>CACHE_SIZE</th><th>LAST_NUMBER</th><th>% UTILIZADO</th><th>CYCLE_FLAG</th></tr>"

|| '<td>' || CACHE_SIZE || '</td>'
|| '<td>' || LAST_NUMBER || '</td>'

ORDER BY TRUNC(LAST_NUMBER/MAX_VALUE*100,2) DESC;

EXIT;
EOF

    echo "</table>" >> "$HTML_FILE"

check_index_invalid() {

        echo "<h3>Índices Inválidos ou Inutilizáveis</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>OWNER</th><th>STATUS</th><th>COUNT</th></tr>"

check_statistcs_outdated() {
    {
        echo "<h3>Estatísticas Desatualizadas (Tabelas e Índices)</h3>"
        echo "<table border='1' cellpadding='5' cellspacing='0'>"
        echo "<tr><th>OWNER</th><th>LAST_ANALYZED_MAIS_ANTIGO</th><th>QTDE_OBJETOS_DESATUALIZADOS</th><th>TIPO_OBJETO</th></tr>"
    } >> "$HTML_FILE"

    sqlplus -s /@$ORACLE_SID >> "$HTML_FILE" <<EOF
SET HEADING OFF
SET FEEDBACK OFF
SET PAGESIZE 0
SET LINESIZE 1000
SET TERMOUT OFF
SET ECHO OFF

SELECT
    '<tr>' ||
    '<td>' || OWNER || '</td>' ||
    '<td>' || NVL(TO_CHAR(LAST_ANALYZED_MAIS_ANTIGO,'DD/MM/YYYY HH24:MI:SS'),'NUNCA ANALISADO') || '</td>' ||
    '<td>' || QTDE_OBJETOS_DESATUALIZADOS || '</td>' ||
    '<td>' || TIPO_OBJETO || '</td>' ||
    '</tr>'
FROM (
    SELECT OWNER,
           MIN(LAST_ANALYZED) AS LAST_ANALYZED_MAIS_ANTIGO,
           COUNT(*) AS QTDE_OBJETOS_DESATUALIZADOS,
           'TABLE' AS TIPO_OBJETO
    FROM DBA_TAB_STATISTICS
    WHERE LAST_ANALYZED IS NULL OR LAST_ANALYZED < SYSDATE - 1
    GROUP BY OWNER
    UNION ALL
    SELECT OWNER,
           MIN(LAST_ANALYZED) AS LAST_ANALYZED_MAIS_ANTIGO,
           COUNT(*) AS QTDE_OBJETOS_DESATUALIZADOS,
           'INDEX' AS TIPO_OBJETO
    FROM DBA_IND_STATISTICS
    WHERE LAST_ANALYZED IS NULL OR LAST_ANALYZED < SYSDATE - 1
    GROUP BY OWNER
    ORDER BY OWNER, TIPO_OBJETO
);

EXIT;
EOF

    echo "</table>" >> "$HTML_FILE"
}
